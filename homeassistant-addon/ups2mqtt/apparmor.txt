# Home Assistant add-on AppArmor profile for ups2mqtt.
#include <tunables/global>

profile ups2mqtt flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  network,
  capability,
  file,

  signal (send) set=(kill,term,int,hup,cont),

  # s6-overlay / Home Assistant base image init path.
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/local/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio is used by the Home Assistant add-on init script.
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Runtime state, Supervisor options, and mapped SSL content.
  /data/** rw,
  /ssl/** r,
}
